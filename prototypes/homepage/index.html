⌘ useDebounce.tsTypeScript
// A little less noise. A little more focus.
export function useDebounce(value, delay) {
  const [debounced, setDebounced] =
    useState(value);

  useEffect(() => {
    const timer = setTimeout(() => {
      setDebounced(value);
    }, delay);
    return () => clearTimeout(timer);
  }, [value, delay]);

  return debounced;
}
